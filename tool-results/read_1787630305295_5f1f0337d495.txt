    97→    97→function parseWonText(text: string): { lotNumber: string; price: string; bodyNumber: string | null }[] {
    98→    98→  const out: { lotNumber: string; price: string; bodyNumber: string | null }[] = [];
    99→    99→  const lines = text.split(/\r?\n/).map((l) => l.trim()).filter(Boolean);
   100→   100→  for (const line of lines) {
   101→   101→    const nums = line.match(/\d{1,3}(?:[,.]\d{3})+|\d+/g) || [];
   102→   102→    if (nums.length === 0) continue;
   103→   103→    const firstNum = nums[0];
   104→   104→    const lastNum = nums[nums.length - 1];
   105→   105→    const lotNumber = firstNum.replace(/[,.]/g, "");
   106→   106→    const price = lastNum.replace(/[,.]/g, "");
   107→   107→    const firstIdx = line.indexOf(firstNum);
   108→   108→    const lastIdx = line.lastIndexOf(lastNum);
   109→   109→    let bodyNumber: string | null = null;
   110→   110→    if (firstIdx !== -1 && lastIdx !== -1 && lastIdx > firstIdx + firstNum.length) {
   111→   111→      bodyNumber = line.substring(firstIdx + firstNum.length, lastIdx).trim() || null;
   112→   112→    }
   113→   113→    out.push({ lotNumber, price, bodyNumber });
   114→   114→  }
   115→   115→  return out;
   116→   116→}
