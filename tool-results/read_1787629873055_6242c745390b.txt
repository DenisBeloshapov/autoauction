    99→    99→export function ClientPanel() {
   100→   100→  const { t } = useLanguage();
   101→   101→  const [activeTab, setActiveTab] = useState("myLots");
   102→   102→  const [lots, setLots] = useState<Lot[]>([]);
   103→   103→  const [wonLots, setWonLots] = useState<WonLot[]>([]);
   104→   104→  const [loading, setLoading] = useState(true);
   105→   105→  const [showAdd, setShowAdd] = useState(false);
   106→   106→  const [lotEntries, setLotEntries] = useState<string[]>([""]);
   107→   107→  const [lotComment, setLotComment] = useState("");
   108→   108→  const [submitting, setSubmitting] = useState(false);
   109→   109→
   110→   110→  // Delivery modal state
   111→   111→  const [deliveryWonLot, setDeliveryWonLot] = useState<WonLot | null>(null);
   112→   112→  const [deliveryMethod, setDeliveryMethod] = useState<Method>("DUTY");
   113→   113→  const [ownerFullName, setOwnerFullName] = useState("");
   114→   114→  const [ownerAddress, setOwnerAddress] = useState("");
   115→   115→
   116→   116→  const token = typeof window !== "undefined" ? localStorage.getItem("auth_token") : null;
   117→   117→  const authHeaders: HeadersInit = token
   118→   118→    ? { Authorization: `Bearer ${token}`, "Content-Type": "application/json" }
   119→   119→    : { "Content-Type": "application/json" };
   120→   120→
   121→   121→  const loadData = useCallback(async () => {
   122→   122→    setLoading(true);
   123→   123→    try {
   124→   124→      const [lotsRes, wonRes] = await Promise.all([
   125→   125→        fetch("/api/lots", { headers: authHeaders }),
   126→   126→        fetch("/api/won-lots", { headers: authHeaders }),
   127→   127→      ]);
   128→   128→      if (lotsRes.ok) {
   129→   129→        const data = await lotsRes.json();
   130→   130→        setLots(data.lots || []);
   131→   131→      }
   132→   132→      if (wonRes.ok) {
   133→   133→        const data = await wonRes.json();
   134→   134→        setWonLots(data.wonLots || []);
   135→   135→      }
   136→   136→    } finally {
   137→   137→      setLoading(false);
   138→   138→    }
   139→   139→  }, [token]);
   140→   140→
   141→   141→  useEffect(() => {
   142→   142→    void loadData();
   143→   143→  }, [loadData]);
   144→   144→
   145→   145→  // Realtime: обновляем данные при WS-событиях
   146→   146→  useRealtime(["lot:created", "lot:updated", "lot:deleted", "wonlot:created", "delivery:created"], () => {
   147→   147→    loadData();
   148→   148→  });

   150→   150→  const tabs: Tab[] = [
   151→   151→    { key: "myLots", label: t("nav.myLots"), icon: <ListChecks className="w-4 h-4" /> },
   152→   152→    { key: "wonLots", label: t("nav.wonLots"), icon: <Trophy className="w-4 h-4" /> },
   153→   153→    { key: "delivery", label: t("nav.delivery"), icon: <Truck className="w-4 h-4" /> },
   154→   154→  ];

   156→   156→  // --- Multiple-lots form helpers ---
   157→   157→  const addLotEntry = () => {
   158→   158→    setLotEntries((prev) => [...prev, ""]);
   159→   159→  };
   160→   160→  const removeLotEntry = (idx: number) => {
   161→   161→    setLotEntries((prev) => prev.filter((_, i) => i !== idx));
   162→   162→  };
   163→   163→  const updateLotEntry = (idx: number, val: string) => {
   164→   164→    setLotEntries((prev) => prev.map((e, i) => (i === idx ? val : e)));
   165→   165→  };
   166→   166→
   167→   167→  // Extract lot number from text (first number sequence)
   168→   168→  const extractLotNumber = (text: string): string | null => {
   169→   169→    const m = text.match(/\d{1,3}(?:,\d{3})+|\d+/);
   170→   170→    return m ? m[0].replace(/,/g, "") : null;
   171→   171→  };
   172→   172→
   173→   173→  // Valid lots = non-empty entries with an extractable lot number
   174→   174→  const validLots = lotEntries
   175→   175→    .map((text, idx) => ({ idx, text, lotNumber: extractLotNumber(text) }))
   176→   176→    .filter((e) => e.text.trim() && e.lotNumber);

   179→   179→    if (validLots.length === 0) return;

   181→   181→    let created = 0;
   182→   182→    let failed = 0;

   184→   184→      for (const entry of validLots) {
   185→   185→        const res = await fetch("/api/lots", {
   186→   186→          method: "POST",
   187→   187→          headers: authHeaders,
   188→   188→          body: JSON.stringify({
   189→   189→            lotText: entry.text,
   190→   190→            comment: lotComment || null,
   191→   191→          }),
   192→   192→        });

   194→   194→          created++;
   195→   195→        } else {
   196→   196→          failed++;

   198→   198→          toast.error(`#${entry.lotNumber}: ${data.error || t("common.error")}`);
   199→   199→        }

   202→   202→        toast.success(t("lots.batchCreated").replace("{n}", String(created)));

   205→   205→        setLotEntries([""]);
   206→   206→        setLotComment("");
   207→   207→        setShowAdd(false);
   208→   208→        await loadData();
   209→   209→      } else {
   210→   210→        // Keep entries that failed for correction
   211→   211→        await loadData();
   212→   212→      }
   213→   213→    } finally {
   214→   214→      setSubmitting(false);
   215→   215→    }
   216→   216→  };

   219→   219→    if (!confirm(t("lots.deleteConfirm"))) return;
   220→   220→    const res = await fetch(`/api/lots?id=${id}`, {
   221→   221→      method: "DELETE",
   222→   222→      headers: authHeaders,
   223→   223→    });
   224→   224→    if (res.ok) {
   225→   225→      toast.success(t("common.success"));
   226→   226→      await loadData();
   227→   227→    } else {
   228→   228→      const data = await res.json().catch(() => ({}));
   229→   229→      toast.error(data.error || t("common.error"));
   230→   230→    }
   231→   231→  };
   232→   232→
   233→   233→  const submitDelivery = async () => {
   234→   234→    if (!deliveryWonLot) return;

   236→   236→      toast.error(t("delivery.ownerFullName") + " / " + t("delivery.ownerAddress"));
   237→   237→      return;
   238→   238→    }
   239→   239→    const res = await fetch("/api/delivery", {
   240→   240→      method: "POST",
   241→   241→      headers: authHeaders,
   242→   242→      body: JSON.stringify({
   243→   243→        wonLotId: deliveryWonLot.id,
   244→   244→        method: deliveryMethod,
   245→   245→        ownerFullName: deliveryMethod === "DUTY" ? ownerFullName : null,
   246→   246→        ownerAddress: deliveryMethod === "DUTY" ? ownerAddress : null,
   247→   247→      }),
   248→   248→    });
   249→   249→    if (res.ok) {
   250→   250→      toast.success(t("common.success"));
   251→   251→      setDeliveryWonLot(null);
   252→   252→      setOwnerFullName("");
   253→   253→      setOwnerAddress("");
   254→   254→      await loadData();
   255→   255→    } else {
   256→   256→      const data = await res.json().catch(() => ({}));
   257→   257→      toast.error(data.error || t("common.error"));
   258→   258→    }
   259→   259→  };
   260→   260→
   261→   261→  if (loading) {
   262→   262→    return (
   263→   263→      <div className="min-h-screen flex items-center justify-center bg-background">
   264→   264→        <Loader2 className="w-8 h-8 animate-spin text-primary" />
   265→   265→      </div>
   266→   266→    );
   267→   267→  }
   268→   268→
   269→   269→  const wonWithDelivery = wonLots.filter((w) => w.deliveryReqs.length > 0);
   270→   270→
   271→   271→  return (
   272→   272→    <AppShell
   273→   273→      tabs={tabs}

   275→   275→      onTabChange={setActiveTab}
   276→   276→      fab={
   277→   277→        activeTab === "myLots"
   278→   278→          ? { label: t("lots.add"), icon: <Plus className="w-5 h-5" />, onClick: () => setShowAdd(true) }
   279→   279→          : undefined
   280→   280→      }
   281→   281→    >
   282→   282→      {activeTab === "myLots" && (
   283→   283→        <div>
   284→   284→          <div className="hidden md:flex items-center justify-between mb-6">
   285→   285→            <h1 className="text-2xl font-bold">{t("nav.myLots")}</h1>
   286→   286→            <button
   287→   287→              onClick={() => setShowAdd(true)}
   288→   288→              className="h-10 px-4 rounded-xl aa-grad text-white text-sm font-semibold flex items-center gap-1.5 hover:brightness-110 transition"
   289→   289→            >
   290→   290→              <Plus className="w-4 h-4" /> {t("lots.add")}
   291→   291→            </button>
   292→   292→          </div>
   293→   293→          <h1 className="md:hidden text-xl font-bold mb-4">{t("nav.myLots")}</h1>
   294→   294→
   295→   295→          {lots.length === 0 ? (
   296→   296→            <EmptyState icon={<Package className="w-10 h-10 text-muted-foreground" />} text={t("lots.noLots")} />
   297→   297→          ) : (

   299→   299→              lots={lots}
   300→   300→              t={t}
   301→   301→              onDelete={deleteLot}
   302→   302→            />
   303→   303→          )}
   304→   304→        </div>
   305→   305→      )}
   306→   306→
   307→   307→      {activeTab === "wonLots" && (
   308→   308→        <div>
   309→   309→          <h1 className="text-2xl font-bold mb-6">{t("nav.wonLots")}</h1>
   310→   310→          {wonLots.length === 0 ? (
   311→   311→            <EmptyState icon={<Trophy className="w-10 h-10 text-muted-foreground" />} text={t("wonLots.noWonLots")} />
   312→   312→          ) : (
   313→   313→            <motion.div
   314→   314→              variants={containerVariants}

   317→   317→              className="aa-card overflow-hidden"
   318→   318→            >

   320→   320→                <table className="w-full text-sm">

   328→   328→                  </thead>
   329→   329→                  <tbody>
   330→   330→                    {wonLots.map((wl) => {
   331→   331→                      const hasDelivery = wl.deliveryReqs.length > 0;
   332→   332→                      return (
   333→   333→                        <motion.tr
   334→   334→                          key={wl.id}
   335→   335→                          variants={itemVariants}

   357→   357→                                  setDeliveryWonLot(wl);
   358→   358→                                  setDeliveryMethod("DUTY");
   359→   359→                                  setOwnerFullName("");
   360→   360→                                  setOwnerAddress("");
   361→   361→                                }}
   362→   362→                                className="text-xs h-8 px-3 rounded-lg border border-border hover:bg-muted hover:border-primary/40 font-medium flex items-center gap-1 transition"

   365→   365→                              </button>
   366→   366→                            )}
   367→   367→                          </td>
   368→   368→                        </motion.tr>
   369→   369→                      );
   370→   370→                    })}
   371→   371→                  </tbody>

   379→   379→      {activeTab === "delivery" && (

   394→   394→                return (

   396→   396→                    key={wl.id}
   397→   397→                    variants={itemVariants}

   399→   399→                  >

   401→   401→                    <div className="flex items-start justify-between p-5 pb-4 gap-3">
   402→   402→                      <div className="flex items-center gap-3 min-w-0">

   404→   404→                          <Truck className="w-5 h-5 text-white" />
   405→   405→                        </div>
   406→   406→                        <div className="min-w-0">
   407→   407→                          <div className="text-[10px] text-muted-foreground uppercase tracking-wider font-semibold">
   408→   408→                            {t("delivery.lotNumber")}
   409→   409→                          </div>
   410→   410→                          <div className="text-xl font-extrabold aa-mono text-primary mt-0.5 truncate">
   411→   411→                            #{wl.lot.lotNumber}
   412→   412→                          </div>
   413→   413→                        </div>
   414→   414→                      </div>
   415→   415→                      <span className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-emerald-50 text-emerald-700 border border-emerald-200 text-xs font-semibold flex-shrink-0">
   416→   416→                        <Check className="w-3.5 h-3.5" />
   417→   417→                        {t("delivery.choosen")}
   418→   418→                      </span>
   419→   419→                    </div>
   420→   420→
   421→   421→                    {/* Body */}
   422→   422→                    <div className="px-5 pb-5 grid grid-cols-1 sm:grid-cols-2 gap-4">
   423→   423→                      {/* Lot info */}
   424→   424→                      <div className="space-y-2.5">

   427→   427→                          {t("delivery.lotInfo")}
   428→   428→                        </div>
   429→   429→                        <InfoRow icon={<Hash className="w-3.5 h-3.5" />} label={t("delivery.lotNumber")} value={`#${wl.lot.lotNumber}`} mono />
   430→   430→                        <InfoRow icon={<Coins className="w-3.5 h-3.5" />} label={t("delivery.price")} value={wl.price ? `${wl.price.toLocaleString()} ${t("wonLots.currency")}` : "—"} mono />

   432→   432→                        <InfoRow icon={<Calendar className="w-3.5 h-3.5" />} label={t("delivery.createdAt")} value={req ? new Date(req.createdAt).toLocaleString() : "—"} />
   433→   433→                        {wl.lot.rawText && (

   437→   437→                            </div>
   438→   438→                            <p className="text-xs text-foreground/80 break-words">
   439→   439→                              {wl.lot.rawText}
   440→   440→                            </p>
   441→   441→                          </div>
   442→   442→                        )}
   443→   443→                      </div>
   444→   444→
   445→   445→                      {/* Owner data (only for DUTY) */}
   446→   446→                      {isDuty && (

   675→   675→  type Group = {
   676→   676→    key: string;
   677→   677→    comment: string | null;

   681→   681→  const groups: Group[] = (() => {

   685→   685→    for (const lot of lots) {
   686→   686→      const c = lot.comment && lot.comment.trim() ? lot.comment.trim() : null;
   687→   687→      if (c) {
   688→   688→        if (!map.has(c)) map.set(c, []);
   689→   689→        map.get(c)!.push(lot);
   690→   690→      } else {
   691→   691→        individual.push(lot);

   700→   700→      out.push({ key: `single:${lot.id}`, comment: null, lots: [lot] });
   701→   701→    }
   702→   702→    return out;
   703→   703→  })();
   704→   704→
