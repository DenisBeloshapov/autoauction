     4→import { motion } from "framer-motion";
     5→import {
     6→  ListChecks,
     7→  Trophy,
     8→  Truck,
     9→  Plus,
    10→  Trash2,
    11→  Package,
    12→  Loader2,
    13→  X,
    14→  Check,
    15→  User,
    16→  MapPin,
    17→  Hash,
    18→  FileText,
    19→  Calendar,
    20→  Coins,
    21→  Boxes,
    22→} from "lucide-react";
    23→import { useAuth } from "@/contexts/AuthContext";
    24→import { useLanguage } from "@/contexts/LanguageContext";
    25→import { AppShell, type Tab } from "./AppShell";
    26→import { Modal } from "./Modal";
    27→import { StatusBadge } from "./StatusBadge";
    28→import { toast } from "sonner";
    29→import { useRealtime } from "@/hooks/use-realtime";
    30→
    31→type Lot = {
    32→  id: string;
    33→  lotNumber: string;
    34→  rawText: string | null;
    35→  comment: string | null;
    36→  status: string;
    37→  createdAt: string;
    38→  client: { id: string; name: string | null; username: string };
    39→  wonLot: {
    40→    id: string;
    41→    price: number | null;
    42→    currency: string;
    43→    status: string;
    44→  } | null;
    45→};
    46→
    47→type WonLot = {
    48→  id: string;
    49→  price: number | null;
    50→  currency: string;
    51→  status: string;
    52→  lot: {
    53→    id: string;
    54→    lotNumber: string;
    55→    rawText: string | null;
    56→    client: { id: string; name: string | null; username: string };
    57→  };
    58→  deliveryReqs: {
    59→    id: string;
    60→    method: string | null;
    61→    status: string;
    62→    ownerFullName: string | null;
    63→    ownerAddress: string | null;
    64→    createdAt: string;
    65→  }[];
    66→};
    67→
    68→const METHODS = [
    69→  "DUTY",
    70→  "DISASSEMBLY",
    71→  "CUT_REAR",
    72→  "CUT_FRONT",
    73→  "CUT_REAR_ARCS",
    74→] as const;
    75→type Method = (typeof METHODS)[number];
    76→
    77→// Stagger animation: cards appear one after another
    78→const containerVariants = {
    79→  hidden: { opacity: 0 },
    80→  visible: {
    81→    opacity: 1,
    82→    transition: {
    83→      staggerChildren: 0.06,
    84→      delayChildren: 0.05,
    85→    },
    86→  },
    87→};
    88→
    89→const itemVariants = {
    90→  hidden: { opacity: 0, y: 18, scale: 0.98 },
    91→  visible: {
    92→    opacity: 1,
    93→    y: 0,
    94→    scale: 1,
    95→    transition: { type: "spring", damping: 22, stiffness: 280 },
    96→  },
    97→};
    98→
    99→export function ClientPanel() {
   100→  const { t } = useLanguage();
   101→  const [activeTab, setActiveTab] = useState("myLots");
   102→  const [lots, setLots] = useState<Lot[]>([]);
   103→  const [wonLots, setWonLots] = useState<WonLot[]>([]);
   104→  const [loading, setLoading] = useState(true);
   105→  const [showAdd, setShowAdd] = useState(false);
   106→  const [lotEntries, setLotEntries] = useState<string[]>([""]);
   107→  const [lotComment, setLotComment] = useState("");
   108→  const [submitting, setSubmitting] = useState(false);
   109→
   110→  // Delivery modal state
   111→  const [deliveryWonLot, setDeliveryWonLot] = useState<WonLot | null>(null);
   112→  const [deliveryMethod, setDeliveryMethod] = useState<Method>("DUTY");
   113→  const [ownerFullName, setOwnerFullName] = useState("");
   114→  const [ownerAddress, setOwnerAddress] = useState("");
   115→
   116→  const token = typeof window !== "undefined" ? localStorage.getItem("auth_token") : null;
   117→  const authHeaders: HeadersInit = token
   118→    ? { Authorization: `Bearer ${token}`, "Content-Type": "application/json" }
   119→    : { "Content-Type": "application/json" };
   120→
   121→  const loadData = useCallback(async () => {
   122→    setLoading(true);
   123→    try {
   124→      const [lotsRes, wonRes] = await Promise.all([
   125→        fetch("/api/lots", { headers: authHeaders }),
   126→        fetch("/api/won-lots", { headers: authHeaders }),
   127→      ]);
   128→      if (lotsRes.ok) {
   129→        const data = await lotsRes.json();
   130→        setLots(data.lots || []);
   131→      }
   132→      if (wonRes.ok) {
   133→        const data = await wonRes.json();
   134→        setWonLots(data.wonLots || []);
   135→      }
   136→    } finally {
   137→      setLoading(false);
   138→    }
   139→  }, [token]);
   140→
   141→  useEffect(() => {
   142→    void loadData();
   143→  }, [loadData]);
   144→
   145→  // Realtime: обновляем данные при WS-событиях
   146→  useRealtime(["lot:created", "lot:updated", "lot:deleted", "wonlot:created", "delivery:created"], () => {
   147→    loadData();
   148→  });
   149→
   150→  const tabs: Tab[] = [
   151→    { key: "myLots", label: t("nav.myLots"), icon: <ListChecks className="w-4 h-4" /> },
   152→    { key: "wonLots", label: t("nav.wonLots"), icon: <Trophy className="w-4 h-4" /> },
   153→    { key: "delivery", label: t("nav.delivery"), icon: <Truck className="w-4 h-4" /> },
   154→  ];
   155→
   156→  // --- Multiple-lots form helpers ---
   157→  const addLotEntry = () => {
   158→    setLotEntries((prev) => [...prev, ""]);
   159→  };
   160→  const removeLotEntry = (idx: number) => {
   161→    setLotEntries((prev) => prev.filter((_, i) => i !== idx));
   162→  };
   163→  const updateLotEntry = (idx: number, val: string) => {
   164→    setLotEntries((prev) => prev.map((e, i) => (i === idx ? val : e)));
   165→  };
   166→
   167→  // Extract lot number from text (first number sequence)
   168→  const extractLotNumber = (text: string): string | null => {
   169→    const m = text.match(/\d{1,3}(?:,\d{3})+|\d+/);
   170→    return m ? m[0].replace(/,/g, "") : null;
   171→  };
   172→
   173→  // Valid lots = non-empty entries with an extractable lot number
   174→  const validLots = lotEntries
   175→    .map((text, idx) => ({ idx, text, lotNumber: extractLotNumber(text) }))
   176→    .filter((e) => e.text.trim() && e.lotNumber);
   177→
   178→  const submitLots = async () => {
   179→    if (validLots.length === 0) return;
   180→    setSubmitting(true);
   181→    let created = 0;
   182→    let failed = 0;
   183→    try {
   184→      for (const entry of validLots) {
   185→        const res = await fetch("/api/lots", {
   186→          method: "POST",
   187→          headers: authHeaders,
   188→          body: JSON.stringify({
   189→            lotText: entry.text,
   190→            comment: lotComment || null,
   191→          }),
   192→        });
   193→        if (res.ok) {
   194→          created++;
   195→        } else {
   196→          failed++;
   197→          const data = await res.json().catch(() => ({}));
   198→          toast.error(`#${entry.lotNumber}: ${data.error || t("common.error")}`);
   199→        }
   200→      }
   201→      if (created > 0) {
   202→        toast.success(t("lots.batchCreated").replace("{n}", String(created)));
   203→      }
   204→      if (failed === 0) {
   205→        setLotEntries([""]);
   206→        setLotComment("");
   207→        setShowAdd(false);
   208→        await loadData();
   209→      } else {
   210→        // Keep entries that failed for correction
   211→        await loadData();
   212→      }
   213→    } finally {
   214→      setSubmitting(false);
   215→    }
   216→  };
   217→
   218→  const deleteLot = async (id: string) => {
   219→    if (!confirm(t("lots.deleteConfirm"))) return;
   220→    const res = await fetch(`/api/lots?id=${id}`, {
   221→      method: "DELETE",
   222→      headers: authHeaders,
   223→    });
   224→    if (res.ok) {
   225→      toast.success(t("common.success"));
   226→      await loadData();
   227→    } else {
   228→      const data = await res.json().catch(() => ({}));
   229→      toast.error(data.error || t("common.error"));
   230→    }
   231→  };
   232→
   233→  const submitDelivery = async () => {
   234→    if (!deliveryWonLot) return;
   235→    if (deliveryMethod === "DUTY" && (!ownerFullName || !ownerAddress)) {
   236→      toast.error(t("delivery.ownerFullName") + " / " + t("delivery.ownerAddress"));
   237→      return;
   238→    }
   239→    const res = await fetch("/api/delivery", {
   240→      method: "POST",
   241→      headers: authHeaders,
   242→      body: JSON.stringify({
   243→        wonLotId: deliveryWonLot.id,
   244→        method: deliveryMethod,
   245→        ownerFullName: deliveryMethod === "DUTY" ? ownerFullName : null,
   246→        ownerAddress: deliveryMethod === "DUTY" ? ownerAddress : null,
   247→      }),
   248→    });
   249→    if (res.ok) {
   250→      toast.success(t("common.success"));
   251→      setDeliveryWonLot(null);
   252→      setOwnerFullName("");
   253→      setOwnerAddress("");
   254→      await loadData();
   255→    } else {
   256→      const data = await res.json().catch(() => ({}));
   257→      toast.error(data.error || t("common.error"));
   258→    }
   259→  };
   260→
   261→  if (loading) {
   262→    return (
   263→      <div className="min-h-screen flex items-center justify-center bg-background">
   264→        <Loader2 className="w-8 h-8 animate-spin text-primary" />
   265→      </div>
   266→    );
   267→  }
   268→
   269→  const wonWithDelivery = wonLots.filter((w) => w.deliveryReqs.length > 0);
   270→
   271→  return (
   272→    <AppShell
   273→      tabs={tabs}
   274→      activeTab={activeTab}
   275→      onTabChange={setActiveTab}
   276→      fab={
   277→        activeTab === "myLots"
   278→          ? { label: t("lots.add"), icon: <Plus className="w-5 h-5" />, onClick: () => setShowAdd(true) }
   279→          : undefined
   280→      }
   281→    >
   282→      {activeTab === "myLots" && (
   283→        <div>
   284→          <div className="hidden md:flex items-center justify-between mb-6">
   285→            <h1 className="text-2xl font-bold">{t("nav.myLots")}</h1>
   286→            <button
   287→              onClick={() => setShowAdd(true)}
   288→              className="h-10 px-4 rounded-xl aa-grad text-white text-sm font-semibold flex items-center gap-1.5 hover:brightness-110 transition"
   289→            >
   290→              <Plus className="w-4 h-4" /> {t("lots.add")}
   291→            </button>
   292→          </div>
   293→          <h1 className="md:hidden text-xl font-bold mb-4">{t("nav.myLots")}</h1>
   294→
   295→          {lots.length === 0 ? (
   296→            <EmptyState icon={<Package className="w-10 h-10 text-muted-foreground" />} text={t("lots.noLots")} />
   297→          ) : (
   298→            <MyLotsGrouped
   299→              lots={lots}
   300→              t={t}
   301→              onDelete={deleteLot}
   302→            />
   303→          )}
   304→        </div>
   305→      )}
   306→
   307→      {activeTab === "wonLots" && (
   308→        <div>
   309→          <h1 className="text-2xl font-bold mb-6">{t("nav.wonLots")}</h1>
   310→          {wonLots.length === 0 ? (
   311→            <EmptyState icon={<Trophy className="w-10 h-10 text-muted-foreground" />} text={t("wonLots.noWonLots")} />
   312→          ) : (
   313→            <motion.div
   314→              variants={containerVariants}
   315→              initial="hidden"
   316→              animate="visible"
   317→              className="aa-card overflow-hidden"
   318→            >
   319→              <div className="overflow-x-auto scroll-slim">
   320→                <table className="w-full text-sm">
   321→                  <thead className="bg-muted/50 text-muted-foreground text-xs uppercase">
   322→                    <tr>
   323→                      <th className="text-left px-4 py-3">{t("lots.lotNumber")}</th>
   324→                      <th className="text-left px-4 py-3 hidden sm:table-cell">{t("lots.rawText")}</th>
   325→                      <th className="text-right px-4 py-3">{t("wonLots.price")}</th>
   326→                      <th className="text-left px-4 py-3">{t("common.actions")}</th>
   327→                    </tr>
   328→                  </thead>
   329→                  <tbody>
   330→                    {wonLots.map((wl) => {
   331→                      const hasDelivery = wl.deliveryReqs.length > 0;
   332→                      return (
   333→                        <motion.tr
   334→                          key={wl.id}
   335→                          variants={itemVariants}
   336→                          className="border-t border-border/60 hover:bg-muted/30 transition"
   337→                        >
   338→                          <td className="px-4 py-3 aa-mono font-bold text-primary">
   339→                            #{wl.lot.lotNumber}
   340→                          </td>
   341→                          <td className="px-4 py-3 text-muted-foreground max-w-xs truncate hidden sm:table-cell">
   342→                            {wl.lot.rawText || "—"}
   343→                          </td>
   344→                          <td className="px-4 py-3 text-right aa-mono font-semibold">
   345→                            {wl.price ? wl.price.toLocaleString() : "—"}{" "}
   346→                            <span className="text-xs text-muted-foreground">{t("wonLots.currency")}</span>
   347→                          </td>
   348→                          <td className="px-4 py-3">
   349→                            {hasDelivery ? (
   350→                              <span className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-emerald-50 text-emerald-700 border border-emerald-200 text-xs font-semibold">
   351→                                <Check className="w-3.5 h-3.5" />
   352→                                {t("delivery.choosen")}
   353→                              </span>
   354→                            ) : (
   355→                              <button
   356→                                onClick={() => {
   357→                                  setDeliveryWonLot(wl);
   358→                                  setDeliveryMethod("DUTY");
   359→                                  setOwnerFullName("");
   360→                                  setOwnerAddress("");
   361→                                }}
   362→                                className="text-xs h-8 px-3 rounded-lg border border-border hover:bg-muted hover:border-primary/40 font-medium flex items-center gap-1 transition"
   363→                              >
   364→                                <Truck className="w-3 h-3" /> {t("delivery.chooseMethod")}
   365→                              </button>
   366→                            )}
   367→                          </td>
   368→                        </motion.tr>
   369→                      );
   370→                    })}
   371→                  </tbody>
   372→                </table>
   373→              </div>
   374→            </motion.div>
   375→          )}
   376→        </div>
   377→      )}
   378→
   379→      {activeTab === "delivery" && (
   380→        <div>
   381→          <h1 className="text-2xl font-bold mb-6">{t("nav.delivery")}</h1>
   382→          {wonWithDelivery.length === 0 ? (
   383→            <EmptyState icon={<Truck className="w-10 h-10 text-muted-foreground" />} text={t("delivery.noRequests")} />
   384→          ) : (
   385→            <motion.div
   386→              variants={containerVariants}
   387→              initial="hidden"
   388→              animate="visible"
   389→              className="space-y-4"
   390→            >
   391→              {wonWithDelivery.map((wl) => {
   392→                const req = wl.deliveryReqs[0];
   393→                const isDuty = req?.method === "DUTY";
   394→                return (
   395→                  <motion.div
   396→                    key={wl.id}
   397→                    variants={itemVariants}
   398→                    className="aa-card aa-card-hover overflow-hidden"
   399→                  >
   400→                    {/* Header */}
   401→                    <div className="flex items-start justify-between p-5 pb-4 gap-3">
   402→                      <div className="flex items-center gap-3 min-w-0">
   403→                        <div className="w-11 h-11 rounded-2xl aa-grad flex items-center justify-center flex-shrink-0">
   404→                          <Truck className="w-5 h-5 text-white" />
   405→                        </div>
   406→                        <div className="min-w-0">
   407→                          <div className="text-[10px] text-muted-foreground uppercase tracking-wider font-semibold">
   408→                            {t("delivery.lotNumber")}
   409→                          </div>
   410→                          <div className="text-xl font-extrabold aa-mono text-primary mt-0.5 truncate">
   411→                            #{wl.lot.lotNumber}
   412→                          </div>
   413→                        </div>
   414→                      </div>
   415→                      <span className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-emerald-50 text-emerald-700 border border-emerald-200 text-xs font-semibold flex-shrink-0">
   416→                        <Check className="w-3.5 h-3.5" />
   417→                        {t("delivery.choosen")}
   418→                      </span>
   419→                    </div>
   420→
   421→                    {/* Body */}
   422→                    <div className="px-5 pb-5 grid grid-cols-1 sm:grid-cols-2 gap-4">
   423→                      {/* Lot info */}
   424→                      <div className="space-y-2.5">
   425→                        <div className="text-[10px] text-muted-foreground uppercase tracking-wider font-semibold mb-1 flex items-center gap-1.5">
   426→                          <FileText className="w-3 h-3" />
   427→                          {t("delivery.lotInfo")}
   428→                        </div>
   429→                        <InfoRow icon={<Hash className="w-3.5 h-3.5" />} label={t("delivery.lotNumber")} value={`#${wl.lot.lotNumber}`} mono />
   430→                        <InfoRow icon={<Coins className="w-3.5 h-3.5" />} label={t("delivery.price")} value={wl.price ? `${wl.price.toLocaleString()} ${t("wonLots.currency")}` : "—"} mono />
   431→                        <InfoRow icon={<Truck className="w-3.5 h-3.5" />} label={t("delivery.method")} value={req?.method ? t(`delivery.${req.method}`) : "—"} />
   432→                        <InfoRow icon={<Calendar className="w-3.5 h-3.5" />} label={t("delivery.createdAt")} value={req ? new Date(req.createdAt).toLocaleString() : "—"} />
   433→                        {wl.lot.rawText && (
   434→                          <div className="pt-2 mt-1 border-t border-border/60">
   435→                            <div className="text-[10px] text-muted-foreground uppercase tracking-wider font-semibold mb-1">
   436→                              {t("delivery.rawText")}
   437→                            </div>
   438→                            <p className="text-xs text-foreground/80 break-words">
   439→                              {wl.lot.rawText}
   440→                            </p>
   441→                          </div>
   442→                        )}
   443→                      </div>
   444→
   445→                      {/* Owner data (only for DUTY) */}
   446→                      {isDuty && (
   447→                        <div className="space-y-2.5 sm:border-l sm:border-border/60 sm:pl-4">
   448→                          <div className="text-[10px] text-muted-foreground uppercase tracking-wider font-semibold mb-1 flex items-center gap-1.5">
   449→                            <User className="w-3 h-3" />
   450→                            {t("delivery.ownerData")}
   451→                          </div>
   452→                          <InfoRow icon={<User className="w-3.5 h-3.5" />} label={t("delivery.ownerFullName")} value={req?.ownerFullName || "—"} />
   453→                          <InfoRow icon={<MapPin className="w-3.5 h-3.5" />} label={t("delivery.ownerAddress")} value={req?.ownerAddress || "—"} />
   454→                        </div>
   455→                      )}
   456→                    </div>
   457→                  </motion.div>
   458→                );
   459→              })}
   460→            </motion.div>
   461→          )}
   462→        </div>
   463→      )}
   464→
   465→      {/* Add Lot Modal — multiple lots + shared comment */}
   466→      <Modal
   467→        open={showAdd}
   468→        onClose={() => setShowAdd(false)}
   469→        title={t("lots.add")}
   470→        size="fullscreen"
   471→        footer={
   472→          <>
   473→            <button
   474→              onClick={() => setShowAdd(false)}
   475→              className="h-10 px-4 rounded-xl border border-border hover:bg-muted text-sm font-medium transition"
   476→            >
   477→              {t("common.cancel")}
   478→            </button>
   479→            <button
   480→              onClick={submitLots}
   481→              disabled={submitting || validLots.length === 0}
   482→              className="h-10 px-4 rounded-xl aa-grad text-white text-sm font-semibold flex items-center gap-1.5 disabled:opacity-60 transition hover:brightness-110"
   483→            >
   484→              {submitting && <Loader2 className="w-4 h-4 animate-spin" />}
   485→              {t("lots.saveN").replace("{n}", String(validLots.length))}
   486→            </button>
   487→          </>
   488→        }
   489→      >
   490→        <div className="space-y-4">
   491→          {/* Lot entries list */}
   492→          <div className="space-y-3">
   493→            {lotEntries.map((entry, idx) => {
   494→              const lotNum = extractLotNumber(entry);
   495→              return (
   496→                <motion.div
   497→                  key={idx}
   498→                  initial={{ opacity: 0, y: 8 }}
   499→                  animate={{ opacity: 1, y: 0 }}
   500→                  transition={{ type: "spring", damping: 24, stiffness: 300 }}
   501→                  className="space-y-1.5"
   502→                >
   503→                  <div className="flex items-center justify-between">
   504→                    <label className="text-xs font-semibold text-muted-foreground uppercase tracking-wide">
   505→                      {t("lots.lotN").replace("{n}", String(idx + 1))}
   506→                    </label>
   507→                    {lotEntries.length > 1 && (
   508→                      <button
   509→                        onClick={() => removeLotEntry(idx)}
   510→                        className="text-xs text-muted-foreground hover:text-destructive flex items-center gap-0.5 transition"
   511→                        aria-label={t("lots.removeLot")}
   512→                      >
   513→                        <X className="w-3 h-3" />
   514→                      </button>
   515→                    )}
   516→                  </div>
   517→                  <div className="flex items-start gap-2">
   518→                    <textarea
   519→                      value={entry}
   520→                      onChange={(e) => updateLotEntry(idx, e.target.value)}
   521→                      placeholder={t("lots.lotTextPlaceholder")}
   522→                      rows={2}
   523→                      autoFocus={idx === 0}
   524→                      className="flex-1 rounded-xl border border-input bg-white p-3 text-sm font-medium focus:outline-none focus:ring-2 focus:ring-primary resize-none"
   525→                    />
   526→                  </div>
   527→                  {entry.trim() && (
   528→                    <p className="text-xs text-muted-foreground">
   529→                      {t("lots.lotNumber")}:{" "}
   530→                      <span className="aa-mono font-bold text-primary">
   531→                        {lotNum ? `#${lotNum}` : "—"}
   532→                      </span>
   533→                    </p>
   534→                  )}

   664→/** Group client lots into "kits" — sets sharing the same non-empty comment. */
   665→function MyLotsGrouped({
   666→  lots,
   667→  t,
   668→  onDelete,
   669→}: {
   670→  lots: Lot[];
   671→  t: (k: string) => string;
   672→  onDelete: (id: string) => void | Promise<void>;
   673→}) {
   674→  // Group by comment (lots without a comment → individual group of size 1 each)
   675→  type Group = {
   676→    key: string;
   677→    comment: string | null;
   678→    lots: Lot[];
   679→  };
   680→
   681→  const groups: Group[] = (() => {
   682→    const map = new Map<string, Lot[]>();
   683→    const individual: Lot[] = [];
   684→
   685→    for (const lot of lots) {
   686→      const c = lot.comment && lot.comment.trim() ? lot.comment.trim() : null;
   687→      if (c) {
   688→        if (!map.has(c)) map.set(c, []);
   689→        map.get(c)!.push(lot);
   690→      } else {
   691→        individual.push(lot);
   692→      }
   693→    }
   694→
   695→    const out: Group[] = [];
   696→    map.forEach((lotsArr, comment) => {
   697→      out.push({ key: `kit:${comment}`, comment, lots: lotsArr });
   698→    });
   699→    for (const lot of individual) {
   700→      out.push({ key: `single:${lot.id}`, comment: null, lots: [lot] });
   701→    }
   702→    return out;
   703→  })();
   704→
   705→  return (
   706→    <motion.div
   707→      variants={containerVariants}
   708→      initial="hidden"
   709→      animate="visible"
   710→      className="space-y-6"
   711→    >
   712→      {groups.map((group) => {
   713→        const isKit = group.lots.length > 1;
   714→        return (
   715→          <motion.div key={group.key} variants={itemVariants} className="space-y-3">
   716→            {/* Kit header */}
   717→            {isKit && (
   718→              <div className="flex items-center gap-2.5 px-2">
   719→                <div className="w-7 h-7 rounded-lg bg-accent text-accent-foreground flex items-center justify-center flex-shrink-0">
   720→                  <Boxes className="w-4 h-4" />
   721→                </div>
   722→                <div className="min-w-0">
   723→                  <div className="text-[10px] text-muted-foreground uppercase tracking-wider font-semibold">
   724→                    {t("lots.kit")}
   725→                  </div>
   726→                  <div className="text-sm font-semibold text-foreground truncate">
   727→                    💬 {group.comment}
   728→                  </div>
   729→                </div>
   730→                <span className="ml-auto text-xs text-muted-foreground aa-mono">
   731→                  {group.lots.length} {t("lots.kitLots")}
   732→                </span>
   733→              </div>
   734→            )}
   735→
   736→            {/* Lots grid */}
   737→            <div className={`grid grid-cols-1 ${group.lots.length > 1 ? "md:grid-cols-2" : ""} gap-4`}>
   738→              {group.lots.map((lot) => (
   739→                <motion.div
   740→                  key={lot.id}
   741→                  layout
   742→                  initial={{ opacity: 0, y: 8 }}
   743→                  animate={{ opacity: 1, y: 0 }}
   744→                  className="aa-card aa-card-hover p-5"
   745→                >
   746→                  <div className="flex items-start justify-between gap-3 mb-3">
   747→                    <div>
   748→                      <div className="text-[10px] text-muted-foreground uppercase tracking-wider font-semibold">
   749→                        {t("lots.lotNumber")}
   750→                      </div>
   751→                      <div className="text-2xl font-extrabold aa-mono text-primary mt-0.5">
   752→                        #{lot.lotNumber}
   753→                      </div>
   754→                    </div>
   755→                    <StatusBadge status={lot.status} label={t(`status.${lot.status}`)} />
   756→                  </div>
   757→                  {lot.rawText && (
   758→                    <p className="text-sm text-foreground mb-2 break-words">
   759→                      {lot.rawText}
   760→                    </p>
   761→                  )}
   762→                  {/* For single lots (no kit) show comment inline */}
   763→                  {!isKit && lot.comment && (
   764→                    <p className="text-xs text-muted-foreground italic bg-muted/40 rounded-lg px-2.5 py-1.5">
   765→                      💬 {lot.comment}
   766→                    </p>
   767→                  )}
   768→                  <div className="flex items-center justify-between mt-4 pt-3 border-t border-border/60">
   769→                    <span className="text-xs text-muted-foreground">
   770→                      {new Date(lot.createdAt).toLocaleDateString()}
   771→                    </span>
   772→                    <button
   773→                      onClick={() => onDelete(lot.id)}
   774→                      className="text-xs text-destructive hover:underline flex items-center gap-1 transition"
   775→                    >
   776→                      <Trash2 className="w-3 h-3" /> {t("common.delete")}
   777→                    </button>
   778→                  </div>
   779→                </motion.div>
   780→              ))}
   781→            </div>
   782→          </motion.div>
   783→        );
   784→      })}
   785→    </motion.div>
   786→  );
   787→}
