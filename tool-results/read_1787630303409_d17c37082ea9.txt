    97→function parseWonText(text: string): { lotNumber: string; price: string; bodyNumber: string | null }[] {
    98→  const out: { lotNumber: string; price: string; bodyNumber: string | null }[] = [];
    99→  const lines = text.split(/\r?\n/).map((l) => l.trim()).filter(Boolean);
   100→  for (const line of lines) {
   101→    const nums = line.match(/\d{1,3}(?:[,.]\d{3})+|\d+/g) || [];
   102→    if (nums.length === 0) continue;
   103→    const firstNum = nums[0];
   104→    const lastNum = nums[nums.length - 1];
   105→    const lotNumber = firstNum.replace(/[,.]/g, "");
   106→    const price = lastNum.replace(/[,.]/g, "");
   107→    const firstIdx = line.indexOf(firstNum);
   108→    const lastIdx = line.lastIndexOf(lastNum);
   109→    let bodyNumber: string | null = null;
   110→    if (firstIdx !== -1 && lastIdx !== -1 && lastIdx > firstIdx + firstNum.length) {
   111→      bodyNumber = line.substring(firstIdx + firstNum.length, lastIdx).trim() || null;
   112→    }
   113→    out.push({ lotNumber, price, bodyNumber });
   114→  }
   115→  return out;
   116→}

   574→function AdminLotsGrouped({ lots, t, selectedIds, onToggleSelect, onToggleSelectAll, onDelete }: {
   575→  lots: Lot[]; t: (k: string) => string; selectedIds: Set<string>;
   576→  onToggleSelect: (id: string) => void; onToggleSelectAll: () => void; onDelete: (id: string) => void | Promise<void>;
   577→}) {
   578→  type Group = { key: string; comment: string | null; clientName: string; lots: Lot[] };
   579→  const groups: Group[] = (() => {
   580→    const map = new Map<string, Lot[]>(); const individual: Lot[] = [];
   581→    for (const lot of lots) { const c = lot.comment?.trim() || null;
   582→      if (c) { const k = `${lot.clientId}::${c}`; if (!map.has(k)) map.set(k, []); map.get(k)!.push(lot); }
   583→      else individual.push(lot); }
   584→    const out: Group[] = [];
   585→    map.forEach((arr) => { out.push({ key: `kit:${arr[0]?.id}`, comment: arr[0]?.comment || null, clientName: arr[0]?.client.name || arr[0]?.client.username || "—", lots: arr }); });
   586→    for (const lot of individual) out.push({ key: `single:${lot.id}`, comment: null, clientName: lot.client.name || lot.client.username, lots: [lot] });
   587→    return out;
   588→  })();
   589→
   590→  const toggleKit = (group: Group) => {
   591→    const all = group.lots.every((l) => selectedIds.has(l.id));
   592→    group.lots.forEach((l) => { if (all ? selectedIds.has(l.id) : !selectedIds.has(l.id)) onToggleSelect(l.id); });
   593→  };
   594→
   595→  return (<>
   596→    <div className="flex items-center gap-3 mb-4 px-2">
   597→      <input type="checkbox" checked={lots.length > 0 && lots.every((l) => selectedIds.has(l.id))} onChange={onToggleSelectAll} className="w-4 h-4 accent-[#3A6BFF]" aria-label={t("common.all")} />
   598→      <span className="text-xs text-muted-foreground">{t("common.all")} ({lots.length})</span>

   600→    <motion.div variants={containerVariants} initial="hidden" animate="visible" className="space-y-4">
   601→      {groups.map((group) => {
   602→        const isKit = group.lots.length > 1;
   603→        const allKit = group.lots.every((l) => selectedIds.has(l.id));
   604→        const someKit = group.lots.some((l) => selectedIds.has(l.id));
   605→        return (
   606→          <motion.div key={group.key} variants={itemVariants} className="aa-card aa-card-hover overflow-hidden">
   607→            <div className="flex items-center gap-3 p-5 pb-3 border-b border-border/60">
   608→              <input type="checkbox" checked={allKit} ref={(el) => { if (el) el.indeterminate = !allKit && someKit; }} onChange={() => toggleKit(group)} className="w-5 h-5 accent-[#3A6BFF] flex-shrink-0 cursor-pointer" />
   609→              <div className="flex-1 min-w-0">
   610→                {isKit ? (<><div className="text-[10px] text-muted-foreground uppercase tracking-wider font-semibold">{t("lots.kit")} · {group.clientName}</div><div className="text-sm font-semibold text-foreground truncate">💬 {group.comment}</div></>) 
   611→                : (<><div className="text-[10px] text-muted-foreground uppercase tracking-wider font-semibold">{group.clientName}</div><div className="text-lg font-extrabold aa-mono text-primary truncate">#{group.lots[0]?.lotNumber}</div></>)}
   612→              </div>
   613→              <span className="text-xs text-muted-foreground aa-mono flex-shrink-0">{group.lots.length} {t("lots.kitLots")}</span>
   614→            </div>
   615→            <div className="p-5 space-y-3">
   616→              {group.lots.map((lot) => (
   617→                <div key={lot.id} className="flex items-start justify-between gap-3 py-2">
   618→                  <div className="flex items-start gap-3 min-w-0 flex-1">
